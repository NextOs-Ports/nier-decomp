cmake_minimum_required(VERSION 3.25)
project(nier_prepare LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(rexglue 0.10 REQUIRED CONFIG)
add_executable(nier_dump_pe dump_pe.cpp)
target_link_libraries(nier_dump_pe PRIVATE rex::runtime)
rexglue_apply_target_settings(nier_dump_pe)
set(REXGLUE_SOURCE "" CACHE PATH "ReXGlue source providing disassembler headers")
if(REXGLUE_SOURCE)
  add_executable(nier_jumptables jumptables/nier_jumptables.cpp)
  target_include_directories(nier_jumptables PRIVATE jumptables "${REXGLUE_SOURCE}/thirdparty/disasm")
  target_link_libraries(nier_jumptables PRIVATE rex::disasm)
endif()

if(NIER_BUILD_PCM_CAPTURE)
  find_package(OpenSSL REQUIRED)
  add_library(nier_pcm_capture SHARED pcm_capture.cpp)
  target_include_directories(nier_pcm_capture PRIVATE $<TARGET_PROPERTY:rex::runtime,INTERFACE_INCLUDE_DIRECTORIES>)
  target_link_libraries(nier_pcm_capture PRIVATE ${CMAKE_DL_LIBS} OpenSSL::Crypto)
endif()

  add_executable(nier_keyboard_probe keyboard_probe.cpp)
  target_link_libraries(nier_keyboard_probe PRIVATE "${NIER_KEYBOARD_RUNTIME}")
  target_include_directories(nier_keyboard_probe PRIVATE $<TARGET_PROPERTY:rex::runtime,INTERFACE_INCLUDE_DIRECTORIES>)
  target_compile_definitions(nier_keyboard_probe PRIVATE $<TARGET_PROPERTY:rex::runtime,INTERFACE_COMPILE_DEFINITIONS>)
  rexglue_apply_target_settings(nier_keyboard_probe)
endif()
