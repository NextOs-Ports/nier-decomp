cmake_minimum_required(VERSION 3.25)
project(nier_prepare LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(rexglue 0.10 REQUIRED CONFIG)
add_executable(nier_dump_pe dump_pe.cpp)
target_link_libraries(nier_dump_pe PRIVATE rex::runtime)
rexglue_apply_target_settings(nier_dump_pe)
set(REXGLUE_SOURCE "" CACHE PATH "ReXGlue source providing disassembler headers")
if(REXGLUE_SOURCE)
  add_executable(nier_jumptables jumptables/nier_jumptables.cpp)
  target_include_directories(nier_jumptables PRIVATE jumptables "${REXGLUE_SOURCE}/thirdparty/disasm")
  target_link_libraries(nier_jumptables PRIVATE rex::disasm)
endif()
add_executable(nier_wait_probe wait_probe.cpp)
target_link_libraries(nier_wait_probe PRIVATE rex::runtime)
rexglue_apply_target_settings(nier_wait_probe)
